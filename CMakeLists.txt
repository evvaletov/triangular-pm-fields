cmake_minimum_required(VERSION 3.18)
project(triangular_pm_fields VERSION 0.1.0 LANGUAGES C Fortran CXX)
include(CTest)
find_library(GMP_LIBRARY NAMES gmp REQUIRED)
find_path(GMP_INCLUDE_DIR NAMES gmp.h REQUIRED)
if(NOT CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "This release is qualified with GNU Fortran (real128 support).")
endif()
add_library(tan SHARED fortran/native/tan_jet.f90 fortran/native/tan_mod.f90
                       fortran/triangle_exact_geometry.c)
set_target_properties(tan PROPERTIES Fortran_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules")
target_include_directories(tan PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/modules>"
                                      PRIVATE "${GMP_INCLUDE_DIR}")
target_compile_options(tan PRIVATE -O3 -fno-fast-math
  "$<$<COMPILE_LANGUAGE:Fortran>:-ffree-line-length-none>")
target_link_libraries(tan PRIVATE "${GMP_LIBRARY}")
add_executable(tan_cli fortran/native/tan_cli.f90)
target_link_libraries(tan_cli PRIVATE tan)
target_compile_options(tan_cli PRIVATE -ffree-line-length-none -fno-fast-math)
add_executable(example_fortran examples/triangle.f90)
target_link_libraries(example_fortran PRIVATE tan)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/cpp/CMakeLists.txt")
  add_subdirectory(cpp)
endif()
if(BUILD_TESTING)
  add_executable(test_tan fortran/native/test_tan.f90)
  target_link_libraries(test_tan PRIVATE tan)
  target_compile_options(test_tan PRIVATE -ffree-line-length-none -fno-fast-math)
  add_executable(test_tan_c fortran/native/test_tan_c.c)
  target_link_libraries(test_tan_c PRIVATE tan m)
  foreach(t test_tan test_tan_c)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/native_test_work/results")
    add_test(NAME ${t} COMMAND ${t})
    set_tests_properties(${t} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/native_test_work")
  endforeach()
  set_tests_properties(test_tan PROPERTIES FIXTURES_SETUP native_cache_files)
  set_tests_properties(test_tan_c PROPERTIES FIXTURES_REQUIRED native_cache_files)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME python_tests COMMAND "${CMAKE_COMMAND}" -E env
    "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}/python"
    "TAN_BUILD_DIR=${CMAKE_CURRENT_BINARY_DIR}"
    OPENBLAS_NUM_THREADS=1 OMP_NUM_THREADS=1
    "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p "test_*.py")
endif()
